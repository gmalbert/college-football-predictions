# Runtime dependencies for the FastAPI backend ONLY.
#
# Deliberately does not include streamlit. The API reads the pipeline's Parquet
# and JSON artifacts and serves them as JSON; it has no Streamlit runtime. The
# one place that used Streamlit — utils/config.py reading st.secrets — now
# imports it lazily, so the whole API path runs without it.
#
#     python -m pip install -r requirements-api.txt
#
# Use requirements-web.txt instead for local development, which adds streamlit
# (the parity harness needs a Streamlit reference server) and the test tooling.

# --- API server -------------------------------------------------------------
# fastapi 0.118 is the newest release that still accepts pydantic v1, which
# cfbd pins (cfbd requires pydantic<2). It also allows the starlette>=0.46 that
# streamlit requires, so requirements-web.txt can install both side by side.
fastapi==0.118.0
uvicorn>=0.30.0

# --- Data and models --------------------------------------------------------
# The API imports utils/, which loads the Parquet artifacts and runs the saved
# sklearn/XGBoost models for the weekly and value-bet pages.
pandas>=2.1.0
numpy>=1.26.0
pyarrow>=15.0.0
scikit-learn==1.9.0
xgboost==3.4.1
joblib>=1.3.0
plotly>=5.18.0

# api/main.py downscales data_files/logo.png before serving it.
Pillow>=10.0.0

# utils/cfbd_client.py backs the Win Probability page (page 7).
cfbd>=4.5.0
requests>=2.31.0
